DROP DATABASE IF EXISTS sql_practice;
CREATE DATABASE sql_practice;
USE sql_practice;
#---------------
CREATE TABLE departments (
    department_id INT PRIMARY KEY,
    department_name VARCHAR(50),
    manager_name VARCHAR(50)
);
#------------------
CREATE TABLE employees (
    employee_id INT PRIMARY KEY,
    employee_name VARCHAR(50),
    salary INT,
    hire_date DATE,
    department_id INT,
    FOREIGN KEY (department_id) REFERENCES departments(department_id)
);
#-------------------
CREATE TABLE customers (
    customer_id INT PRIMARY KEY,
    customer_name VARCHAR(50),
    region VARCHAR(50),
    signup_date DATE
);
#----------------
CREATE TABLE products (
    product_id INT PRIMARY KEY,
    product_name VARCHAR(50),
    category VARCHAR(50),
    price DECIMAL(10,2),
    cost DECIMAL(10,2)
);
#------------------
CREATE TABLE orders (
    order_id INT PRIMARY KEY,
    customer_id INT,
    employee_id INT,
    order_date DATE,
    FOREIGN KEY (customer_id) REFERENCES customers(customer_id),
    FOREIGN KEY (employee_id) REFERENCES employees(employee_id)
);
#----------------
CREATE TABLE order_items (
    order_item_id INT PRIMARY KEY,
    order_id INT,
    product_id INT,
    quantity INT,
    FOREIGN KEY (order_id) REFERENCES orders(order_id),
    FOREIGN KEY (product_id) REFERENCES products(product_id)
);

# Practice Query # 1:
Select o.order_id,o.order_date ,c.customer_name 
from orders o
join customers c 
 on o.customer_id =c.customer_id;

 # -------query # 2 : 
 # total customers
 select count(*) as  total_customers from customers c;

 # query 3 : distinct regions
 select count(distinct region) as total_regions from customers c;

# query 4 : total sales revenue
select sum(oi.quantity*p.price) as total_revenue 
from order_items oi 
join  products p 
on oi.product_id=p.product_id;

# Query 5 : avg product price per category
select avg(p.price) as avg_price_prod,p.category
from products p 
group by p.category;

# query 6 : total sales revenue per region
select c.region,sum(oi.quantity*p.price)as total_revenue 
from order_items oi 
join products p on p.product_id=oi.product_id
join  orders o on oi.order_id=o.order_id
join customers c on o.customer_id=c.customer_id
group by c.region;

# query 7 : max salary
select max(e.salary) as max_salary from employees e;

# query 8 : employee with max salary
select e.employee_id, e.employee_name,e.salary
from employees e 
where e.salary=(select max(ep.salary) as max_sal from employees ep);

# query 9 : department where total_salary > 150000
select d.department_name , d.department_id , sum(e.salary) as total_salary
from departments d 
join employees e on d.department_id=e.department_id
group by d.department_name, d.department_id
having total_salary  > 150000;

# query 10 :  employee with third highest salary
select e.employee_name, e.employee_id,e.salary 
from employees e 
order by e.salary desc
limit 1 offset 2;

# query 11: departments that have more than 2 employees
select d.department_name , d.department_id , count(e.employee_id) as emp_count
from departments d 
join employees e on d.department_id=e.department_id
group by d.department_name, d.department_id 
having emp_count > 2;
